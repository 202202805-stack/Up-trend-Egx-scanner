def scan_ticker(ticker):
    try:
        # بيانات 10 سنوات
        data_daily = yf.download(ticker, period="10y", interval="1d", progress=False, threads=False)
        if data_daily.empty or len(data_daily) < 100: return None
        if isinstance(data_daily.columns, pd.MultiIndex): data_daily.columns = data_daily.columns.get_level_values(0)

        logic = {'Open': 'first', 'High': 'max', 'Low': 'min', 'Close': 'last', 'Volume': 'sum'}
        data_weekly = data_daily.resample('W').apply(logic).dropna()
        w_rsi = talib.RSI(data_weekly['Close'], 14)
        w_ma50 = talib.SMA(data_weekly['Close'], 50)

        close = data_daily['Close'].ffill()
        high = data_daily['High'].ffill()
        low = data_daily['Low'].ffill()
        open_p = data_daily['Open'].ffill()
        volume = data_daily['Volume'].ffill()

        rsi = talib.RSI(close, 14)
        macd_l, macd_s, _ = talib.MACD(close, 6, 13, 5)
        adx = talib.ADX(high, low, close, 14)
        ema5 = talib.EMA(close, 5)
        mfi = talib.MFI(high, low, close, volume, 14)
        v_ma10 = volume.rolling(10).mean()
        
        ma50_d = close.rolling(50).mean()
        ema9_d = close.ewm(span=9, adjust=False).mean()
        ma20_d = close.rolling(20).mean()
        high_10_d = high.shift(1).rolling(window=10).max()
        slope_d = close.rolling(20).apply(lambda x: np.polyfit(range(len(x)), x, 1)[0], raw=True)

        k_sig = (close > (talib.SMA(close, 10) + talib.TRANGE(high, low, close) * 1.5)).astype(int)
        a_sig = ((close > talib.SMA(close, 8)) & (close > talib.SMA(close, 5))).astype(int)
        sig_count = k_sig + a_sig + ((close - close.shift(7)) > 0).astype(int) + (volume > v_ma10).astype(int)

        def get_strength_score(idx):
            score = 0.0
            if (macd_l.iloc[idx] - macd_s.iloc[idx]) > 0: score += 2.0
            if close.iloc[idx] > ema5.iloc[idx]: score += 1.0
            if 42 <= rsi.iloc[idx] <= 65: score += 2.0
            if sig_count.iloc[idx] >= 3: score += 2.0
            if mfi.iloc[idx] >= 50: score += 1.0
            if volume.iloc[idx] > (v_ma10.iloc[idx] * 1.25): score += 1.0
            return score

        signals = []
        last_success_exit_date = None
        i = 80 
        while i < len(data_daily):
            curr_date = data_daily.index[i]
            if last_success_exit_date is not None:
                if (curr_date - last_success_exit_date).days < 14:
                    i += 1
                    continue

            try:
                last_w_date = data_weekly.index[data_weekly.index <= curr_date][-1]
                is_weekly_strong = (close.iloc[i] > w_ma50.loc[last_w_date]) and (w_rsi.loc[last_w_date] >= 50)
            except: is_weekly_strong = False

            if is_weekly_strong:
                s_today = get_strength_score(i)
                s_yesterday = get_strength_score(i-1)
                candle_range = high.iloc[i] - low.iloc[i]
                candle_body = abs(close.iloc[i] - open_p.iloc[i])
                is_quality = candle_body > (candle_range * 0.6) if candle_range > 0 else False

                if (close.iloc[i] > open_p.iloc[i]) and (s_today >= 7.0) and (s_yesterday < 6.0) and is_quality:
                    if all([close.iloc[i] > ma50_d.iloc[i], ema9_d.iloc[i] > ma20_d.iloc[i], 
                            close.iloc[i] > high_10_d.iloc[i], adx.iloc[i] > 25, slope_d.iloc[i] > 0]):
                        
                        entry_p = float(close.iloc[i])
                        atr = talib.TRANGE(high, low, close).rolling(14).mean().iloc[i]
                        stop_l = round(entry_p - (1.6 * atr), 2)
                        target_p = entry_p * 1.05
                        
                        risk_amt = CAPITAL * RISK_PER_TRADE
                        risk_per_share = entry_p - stop_l
                        pos_size = int(risk_amt / risk_per_share) if risk_per_share > 0 else 0

                        status, days, max_r = "OPEN 🟢", "-", entry_p
                        exit_idx = i
                        for j in range(i + 1, len(data_daily)):
                            max_r = max(max_r, high.iloc[j])
                            if high.iloc[j] >= target_p:
                                status, days = "CLOSED (5% Target) 🛡️", j - i
                                last_success_exit_date = data_daily.index[j]
                                exit_idx = j
                                break
                            if low.iloc[j] <= stop_l:
                                status, days = "STOP LOSS ❌", j - i
                                last_success_exit_date = None
                                exit_idx = j
                                break
                            exit_idx = j

                        signals.append({
                            'Date': curr_date.strftime('%Y-%m-%d'),
                            'Ticker': ticker,
                            'Entry': round(entry_p, 2),
                            'StopLoss': stop_l,
                            'Qty': pos_size,
                            'Score': s_today,
                            'Status': status,
                            'Days': days,
                            'MaxReach': round(max_r, 2)
                        })
                        i = exit_idx
            i += 1
        return signals
    except: return None
