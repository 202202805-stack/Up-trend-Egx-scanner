# --- دالة حساب المؤشرات بدون مكتبات خارجية متخصصة ---
def calc_rsi(series, period=14):
    delta = series.diff()
    gain = (delta.where(delta > 0, 0)).rolling(window=period).mean()
    loss = (-delta.where(delta < 0, 0)).rolling(window=period).mean()
    rs = gain / loss
    return 100 - (100 / (1 + rs))


def calc_ema(series, span):
    return series.ewm(span=span, adjust=False).mean()


def calc_sma(series, window):
    return series.rolling(window=window).mean()


def calc_trange(high, low, close):
    prev_close = close.shift(1)
    tr1 = high - low
    tr2 = (high - prev_close).abs()
    tr3 = (low - prev_close).abs()
    return pd.concat([tr1, tr2, tr3], axis=1).max(axis=1)


def calc_macd(close, fast=6, slow=13, signal=5):
    ema_fast = calc_ema(close, fast)
    ema_slow = calc_ema(close, slow)
    macd_line = ema_fast - ema_slow
    signal_line = calc_ema(macd_line, signal)
    return macd_line, signal_line


# --- دالة سحب البيانات اللحظية من TradingView ---
def fetch_tradingview_live_data() -> pd.DataFrame:
    url = "https://scanner.tradingview.com/egypt/scan"
    payload = {
        "filter": [{"left": "name", "operation": "nempty"}],
        "options": {"active_symbols_only": True},
        "columns": ["name", "description", "open", "high", "low", "close"],
        "sort": {"sortBy": "name", "sortOrder": "asc"},
        "range": [0, 300],
    }
    headers = {"User-Agent": "Mozilla/5.0"}

    try:
        res = requests.post(url, json=payload, headers=headers, timeout=10)
        data = res.json().get("data", [])
    except Exception as e:
        print(f"⚠️ Warning: TradingView fetch failed: {e}")
        return pd.DataFrame()

    tv_rows = []
    for item in data:
        sym = item["s"].replace("EGX:", "")
        d = item["d"]
        open_p = d[2] if d[2] is not None else 0.0
        high_p = d[3] if d[3] is not None else open_p
        low_p = d[4] if d[4] is not None else open_p
        close_p = d[5] if d[5] is not None else open_p

        tv_rows.append({
            "Symbol": f"{sym}.CA",
            "Open": round(float(open_p), 3),
            "High": round(float(high_p), 3),
            "Low": round(float(low_p), 3),
            "Close": round(float(close_p), 3),
        })
    return pd.DataFrame(tv_rows)


# --- دالة دمج شمعة TradingView مع بيانات Yahoo Finance ---
def merge_live_candle(
    df_yf: pd.DataFrame, ticker: str, df_tv_live: pd.DataFrame
) -> pd.DataFrame:
    if df_tv_live.empty or ticker not in df_tv_live["Symbol"].values:
        return df_yf

    tv_row = df_tv_live[df_tv_live["Symbol"] == ticker].iloc[0]
    today_date = pd.Timestamp.now().normalize()

    df = df_yf.copy()

    # إذا كانت آخر شمعة في Yahoo أقدم من اليوم، أضف شمعة اليوم
    if df.index[-1].date() < today_date.date():
        new_row = pd.DataFrame(
            {
                "Open": [tv_row["Open"]],
                "High": [tv_row["High"]],
                "Low": [tv_row["Low"]],
                "Close": [tv_row["Close"]],
                "Volume": [0],
            },
            index=[today_date],
        )
        df = pd.concat([df, new_row])
    else:
        # تحديث شمعة اليوم بأسعار TradingView اللحظية
        df.loc[df.index[-1], "High"] = max(
            df.iloc[-1]["High"], float(tv_row["High"])
        )
        df.loc[df.index[-1], "Low"] = min(
            df.iloc[-1]["Low"], float(tv_row["Low"])
        )
        df.loc[df.index[-1], "Close"] = float(tv_row["Close"])

    return df


class MarketCondition(Enum):
    BULL = "Bull"
    BEAR = "Bear"


class PipeQuality(Enum):
    EXCELLENT = "Excellent"
    GOOD = "Good"
    MODERATE = "Moderate"
    WEAK = "Weak"


@dataclass
class PipeBottomSignal:
    ticker: str
    entry_date: pd.Timestamp
    pattern_confirmation_date: pd.Timestamp
    left_spike_date: pd.Timestamp
    right_spike_date: pd.Timestamp

    entry_price: float
    stop_loss: float
    target_price: float
    target_percent: float
    risk_reward_ratio: float

    left_spike_low: float
    right_spike_low: float
    left_spike_high: float
    right_spike_high: float
    highest_high: float
    lowest_low: float
    pattern_height: float
    prior_drop_pct: float

    market_condition: MarketCondition
    quality_rating: PipeQuality

    # Trade Outcome Tracking
    trade_status: str = "OPEN"  # WIN, LOSS, OPEN
    exit_date: Optional[pd.Timestamp] = None
    exit_price: Optional[float] = None
    pnl_percent: float = 0.0
    holding_weeks: int = 0
    current_price: float = 0.0

    def to_dict(self) -> Dict:
        return {
            "Stock Name": self.ticker,
            "Entry Date": self.entry_date.strftime("%Y-%m-%d"),
            "Entry Price": self.entry_price,
            "Target": self.target_price,
            "Stop Loss": self.stop_loss,
            "Exit Date": (
                self.exit_date.strftime("%Y-%m-%d") if self.exit_date else ""
            ),
            "Current Price": self.current_price,
        }


def generate_buy_signals(data):
    """توليد إشارات الشراء بناءً على المؤشرات الفنية ومنطق الاستراتيجية."""
    df = data.copy()
    if isinstance(df.columns, pd.MultiIndex):
        df.columns = df.columns.get_level_values(0)

    close = df["Close"].ffill()
    open_p = df["Open"].ffill()
    high = df["High"].ffill()
    low = df["Low"].ffill()

    rsi = calc_rsi(close, 14)
    macd_l, macd_s = calc_macd(close, 6, 13, 5)
    ema5 = calc_ema(close, 5)

    trange = calc_trange(high, low, close)
    k_sig = (close > (calc_sma(close, 10) + trange * 1.5)).astype(int)
    a_sig = (
        (close > calc_sma(close, 8))
        & (close > calc_sma(close, 5))
        & (close > calc_sma(close, 3))
    ).astype(int)
    m_sig = ((close - close.shift(7)) > 0).astype(int)
    sig_count = k_sig + a_sig + m_sig

    def calculate_strength(idx):
        score = 0.0
        if macd_l.iloc[idx] > macd_s.iloc[idx]:
            score += 2.0
        if close.iloc[idx] > ema5.iloc[idx]:
            score += 1.0
        if rsi.iloc[idx] > 50:
            score += 2.0
        if sig_count.iloc[idx] >= 2:
            score += 2.0
        return min(score, 10.0)

    signal_flags = []
    strength_values = []

    for i in range(len(df)):
        if i < 2:
            signal_flags.append(False)
            strength_values.append(0.0)
            continue

        current_strength = calculate_strength(i)
        today_ok = (
            (close.iloc[i] > open_p.iloc[i])
            and (current_strength >= 5.0)
            and (macd_l.iloc[i] > macd_s.iloc[i])
        )

        prev_strength = calculate_strength(i - 1)
        yesterday_was_positive = (close.iloc[i - 1] > open_p.iloc[i - 1]) and (
            prev_strength >= 5.0
        )

        if today_ok and not yesterday_was_positive:
            signal_flags.append(True)
            strength_values.append(current_strength)
        else:
            signal_flags.append(False)
            strength_values.append(0.0)

    df["Buy_Signal"] = signal_flags
    df["Signal_Strength"] = strength_values

    return df


class PipeBottomDetector:
    SPIKE_COMPARISON_PERIOD = 52
    MIN_OVERLAP_RATIO = 0.30
    MIN_SPIKE_MULTIPLIER = 1.2
    MAX_CONFIRMATION_WAIT = 8

    def __init__(
        self,
        ticker: str,
        daily_data: pd.DataFrame,
        target_multiplier: float = 0.68,
        min_downtrend_pct: float = 0.15,
        stop_buffer_pct: float = 0.015,
    ):
        self.ticker = ticker
        self.target_multiplier = target_multiplier
        self.min_downtrend_pct = min_downtrend_pct
        self.stop_buffer_pct = stop_buffer_pct
        self.daily_data = daily_data.copy().sort_index()
        self.data = self._resample_to_egx_weekly(daily_data)
        self._prepare_indicators()

    def _resample_to_egx_weekly(self, df: pd.DataFrame) -> pd.DataFrame:
        weekly = (
            df.resample("W-SUN")
            .agg({
                "Open": "first",
                "High": "max",
                "Low": "min",
                "Close": "last",
                "Volume": "sum",
            })
            .dropna()
        )
        return weekly

    def _prepare_indicators(self):
        df = self.data
        df["DownSpike"] = np.minimum(df["Open"], df["Close"]) - df["Low"]
        df["BodySize"] = (df["Close"] - df["Open"]).abs()
        df["BodySizePct"] = df["BodySize"] / df["Close"]
        df["AvgDownSpike"] = (
            df["DownSpike"]
            .rolling(window=self.SPIKE_COMPARISON_PERIOD, min_periods=10)
            .mean()
        )
        df["Range"] = df["High"] - df["Low"]
        df["AvgVolume"] = df["Volume"].rolling(window=20, min_periods=5).mean()

    def _get_market_condition(self, date: pd.Timestamp) -> MarketCondition:
        try:
            weekly_sub = self.data.loc[:date]
            if weekly_sub.empty:
                return MarketCondition.BULL
            price = weekly_sub.iloc[-1]["Close"]
            ma50 = (
                weekly_sub["Close"].rolling(50, min_periods=10).mean().iloc[-1]
                if len(weekly_sub) >= 10
                else price
            )
            return (
                MarketCondition.BULL if price >= ma50 else MarketCondition.BEAR
            )
        except:
            return MarketCondition.BULL

    def _check_spike_pair(self, idx1: int, idx2: int) -> bool:
        df = self.data
        w1, w2 = df.iloc[idx1], df.iloc[idx2]

        avg_spike1, avg_spike2 = w1["AvgDownSpike"], w2["AvgDownSpike"]
        if pd.isna(avg_spike1) or avg_spike1 <= 0:
            return False

        spike1, spike2 = w1["DownSpike"], w2["DownSpike"]
        if (
            spike1 < avg_spike1 * self.MIN_SPIKE_MULTIPLIER
            or spike2 < avg_spike2 * self.MIN_SPIKE_MULTIPLIER
        ):
            return False

        if (w1["DownSpike"] / w1["Close"]) < 0.025 or (
            w2["DownSpike"] / w2["Close"]
        ) < 0.025:
            return False

        body1 = w1["BodySize"]
        body2 = w2["BodySize"]
        max_body = max(body1, body2)
        min_body = min(body1, body2)

        if max_body > 0:
            body_ratio = min_body / max_body
            if body_ratio < 0.35:
                return False

        open_diff = abs(w1["Open"] - w2["Open"]) / w1["Close"]
        close_diff = abs(w1["Close"] - w2["Close"]) / w1["Close"]
        if open_diff > 0.08 or close_diff > 0.08:
            return False

        overlap_high = min(w1["High"], w2["High"])
        overlap_low = max(w1["Low"], w2["Low"])
        overlap = max(0, overlap_high - overlap_low)

        min_range = min(w1["Range"], w2["Range"])
        if min_range <= 0 or (overlap / min_range) < self.MIN_OVERLAP_RATIO:
            return False

        vol1, vol2 = w1["Volume"], w2["Volume"]
        avg_vol1 = w1["AvgVolume"]
        if pd.isna(avg_vol1) or avg_vol1 <= 0:
            return False

        if not (vol1 > vol2 and vol1 > avg_vol1):
            return False

        return True

    def _check_prior_downtrend(self, left_idx: int) -> Tuple[bool, float]:
        df = self.data
        if left_idx < 4:
            return False, 0.0

        lookback = max(0, left_idx - 12)
        recent_high = df.iloc[lookback:left_idx]["High"].max()
        pipe_low = df.iloc[left_idx]["Low"]

        drop_pct = (recent_high - pipe_low) / recent_high
        return drop_pct >= self.min_downtrend_pct, drop_pct

    def _evaluate_trade_outcome(
        self,
        confirmation_date: pd.Timestamp,
        entry_price: float,
        target_price: float,
        stop_loss: float,
    ) -> Tuple[
        str, Optional[pd.Timestamp], Optional[float], float, int, pd.Timestamp
    ]:
        future_daily = self.daily_data.loc[
            self.daily_data.index > confirmation_date
        ]
        days_held = 0

        for idx, row in future_daily.iterrows():
            days_held += 1
            if row["Low"] <= stop_loss:
                pnl = (stop_loss - entry_price) / entry_price
                holding_weeks = max(1, days_held // 5)
                return "LOSS", idx, stop_loss, pnl, holding_weeks, idx
            if row["High"] >= target_price:
                pnl = (target_price - entry_price) / entry_price
                holding_weeks = max(1, days_held // 5)
                return "WIN", idx, target_price, pnl, holding_weeks, idx

        # إذا كانت الصفقة لسه مفتوحة
        last_price = self.daily_data.iloc[-1]["Close"]
        pnl = (last_price - entry_price) / entry_price
        holding_weeks = max(1, len(future_daily) // 5)
        return (
            "OPEN",
            None,
            last_price,
            pnl,
            holding_weeks,
            self.daily_data.index[-1],
        )

    def detect(self) -> List[PipeBottomSignal]:
        signals = []
        df = self.data
        if len(df) < 15:
            return signals

        last_trade_exit_date = pd.Timestamp.min
        latest_current_price = float(self.daily_data.iloc[-1]["Close"])

        for i in range(10, len(df) - 1):
            left_idx, right_idx = i, i + 1
            right_spike_date = df.index[right_idx]

            if right_spike_date <= last_trade_exit_date:
                continue

            if not self._check_spike_pair(left_idx, right_idx):
                continue

            has_downtrend, drop_pct = self._check_prior_downtrend(left_idx)
            if not has_downtrend:
                continue

            l_hi, r_hi = df.iloc[left_idx]["High"], df.iloc[right_idx]["High"]
            l_lo, r_lo = df.iloc[left_idx]["Low"], df.iloc[right_idx]["Low"]

            pipe_high = max(l_hi, r_hi)
            pipe_low = min(l_lo, r_lo)
            pipe_height = pipe_high - pipe_low

            max_end_date = right_spike_date + timedelta(
                weeks=self.MAX_CONFIRMATION_WAIT
            )
            start_history_date = right_spike_date - timedelta(days=60)
            sub_daily = self.daily_data.loc[
                (self.daily_data.index >= start_history_date)
                & (self.daily_data.index <= max_end_date)
            ].copy()

            if len(sub_daily) < 15:
                continue

            sub_daily_with_signals = generate_buy_signals(sub_daily)
            post_pattern_daily = sub_daily_with_signals.loc[
                sub_daily_with_signals.index > right_spike_date
            ]

            confirmed = False
            confirmation_date = None

            for d_idx, d_row in post_pattern_daily.iterrows():
                if d_row["Buy_Signal"] == True:
                    confirmed = True
                    confirmation_date = d_idx
                    break

            if not confirmed:
                continue

            if confirmation_date <= last_trade_exit_date:
                continue

            entry_price = self.daily_data.loc[confirmation_date, "Close"]
            target_price = round(
                entry_price + (pipe_height * self.target_multiplier), 2
            )
            stop_loss = round(pipe_low * (1.0 - self.stop_buffer_pct), 2)

            target_pct = (
                (target_price - entry_price) / entry_price
                if entry_price > 0
                else 0.0
            )

            risk = entry_price - stop_loss
            reward = target_price - entry_price
            rr_ratio = reward / risk if risk > 0 else 0.0

            market_cond = self._get_market_condition(confirmation_date)

            (
                status,
                exit_date,
                exit_price,
                pnl_pct,
                holding_weeks,
                exit_dt,
            ) = self._evaluate_trade_outcome(
                confirmation_date, entry_price, target_price, stop_loss
            )

            last_trade_exit_date = exit_dt

            signal = PipeBottomSignal(
                ticker=self.ticker,
                entry_date=confirmation_date,
                pattern_confirmation_date=confirmation_date,
                left_spike_date=df.index[left_idx],
                right_spike_date=df.index[right_idx],
                entry_price=entry_price,
                stop_loss=stop_loss,
                target_price=target_price,
                target_percent=target_pct,
                risk_reward_ratio=rr_ratio,
                left_spike_low=l_lo,
                right_spike_low=r_lo,
                left_spike_high=l_hi,
                right_spike_high=r_hi,
                highest_high=pipe_high,
                lowest_low=pipe_low,
                pattern_height=pipe_height,
                prior_drop_pct=drop_pct,
                market_condition=market_cond,
                quality_rating=(
                    PipeQuality.EXCELLENT
                    if drop_pct > 0.20
                    else PipeQuality.GOOD
                ),
                trade_status=status,
                exit_date=exit_date,
                exit_price=exit_price,
                pnl_percent=pnl_pct,
                holding_weeks=holding_weeks,
                current_price=latest_current_price,
            )

            signals.append(signal)

        return signals

end_date = datetime.now()
start_date = end_date - timedelta(days=365 + 60)

all_signals_dicts = []

for ticker in tickers:
    try:
        df = yf.download(
            ticker,
            start=start_date.strftime("%Y-%m-%d"),
            end=end_date.strftime("%Y-%m-%d"),
            interval="1d",
            progress=False,
        )
        if df.empty or len(df) < 50:
            continue

        if isinstance(df.columns, pd.MultiIndex):
            df = (
                df.xs(ticker, axis=1, level=1)
                if ticker in df.columns.levels[1]
                else df.droplevel(1, axis=1)
            )

        cols = ["Open", "High", "Low", "Close", "Volume"]
        if not all(c in df.columns for c in cols):
            continue

        df = df[cols].dropna()
        df = merge_live_candle(df, ticker, df_tv_live)

        detector = PipeBottomDetector(
            ticker=ticker,
            daily_data=df,
            target_multiplier=0.68,
            min_downtrend_pct=0.15,
        )
        signals = detector.detect()

        for sig in signals:
            all_signals_dicts.append(sig.to_dict())
    except Exception as e:
        pass

if all_signals_dicts:
    df_results = pd.DataFrame(all_signals_dicts)
    df_results = df_results.drop_duplicates(subset=["Stock Name", "Entry Date"])
    df_results["Entry Date"] = pd.to_datetime(df_results["Entry Date"])
    df_results = df_results.sort_values(by="Entry Date", ascending=True).reset_index(
